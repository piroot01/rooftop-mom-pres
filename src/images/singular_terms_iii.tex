        \def\dx{1.8}
        \def\dy{1.2}
        \def\cx{2}
        \def\cy{0}

        \newcommand{\drawcross}[3]{%
            % draw with color (#1), shift in z by #2
            \begin{scope}[shift={(#1,#2,#3)}]
                \draw[thick,->] (0,0,0) -- (1.5,0,0) node[anchor=north]{$\UV{x}$};
                \draw[thick,->] (0,0,0) -- (0,1.5,0) node[anchor=west]{$\UV{y}$};
                \draw[thick,->] (0,0,0) -- (0,0,1.5) node[anchor=south]{$\UV{z}$};
            \end{scope}
        }

        \def\roofh{1.2}

        \newcommand{\drawroofshapex}[3]{%
            % draw with color (#1), shift in z by #2
            \begin{scope}[shift={(#2,#3,0)}]
                \coordinate (A) at (0,0,0);
                \coordinate (B) at (\dx,0,\roofh);
                \coordinate (C) at (\dx,\dy,\roofh);
                \coordinate (D) at (0,\dy,0);
                \coordinate (E) at (2*\dx,0,0);
                \coordinate (F) at (2*\dx,\dy,0);
                \coordinate (G) at (\dx,0,0);
                \coordinate (H) at (\dx,\dy,0);
                \draw[#1,thick] (A) -- (B) -- (C) -- (D) -- cycle;
                \draw[#1,thick] (B) -- (E) -- (F) -- (C) -- cycle;
                \draw[#1,thick] (A) -- (E) -- (F) -- (D) -- cycle;
                \draw[#1,thick] (G) -- (H);
                \draw[#1,thick] (B) -- (G);
                \draw[#1,thick] (C) -- (H);
            \end{scope}
          }

        \newcommand{\drawroofshapey}[3]{%
            \begin{scope}[shift={(#2,#3,0)}]
                \coordinate (A) at (0,0,0);
                \coordinate (B) at (\dx,0,\roofh);
                \coordinate (C) at (\dx,\dy,\roofh);
                \coordinate (D) at (0,\dy,0);
                \coordinate (E) at (2*\dx,0,0);
                \coordinate (F) at (2*\dx,\dy,0);
                \coordinate (G) at (\dx,0,0);
                \coordinate (H) at (\dx,\dy,0);
                \draw[#1,thick, dashed] (A) -- (B) -- (C) -- (D) -- cycle;
                \draw[#1,thick, dashed] (B) -- (E) -- (F) -- (C) -- cycle;
                \draw[#1,thick, dashed] (A) -- (E) -- (F) -- (D) -- cycle;
                \draw[#1,thick, dashed] (G) -- (H);
                \draw[#1,thick, dashed] (B) -- (G);
                \draw[#1,thick, dashed] (C) -- (H);
            \end{scope}
        }

        \draw[fill = lightgray, opacity = 0.8] (0, 0, 0) -- (2*\dx, 0, 0) -- (2*\dx, \dy, 0) -- (0, \dy, 0) -- cycle;

        \foreach \x in {0,\dx,...,\cx} {
            \foreach \y in {0} {
                \draw[thick] (\x,\y, 0) -- (\x+\dx,\y, 0) -- (\x+\dx, \y+\dy, 0) -- (\x, \y+\dy, 0) -- cycle;
                \draw[thick] (\x,\y, 0) -- (\x+\dx,\y, 0) -- (\x+\dx, \y+\dy, 0) -- (\x, \y+\dy, 0) -- cycle;
            }
        }

        \foreach \x in {0,\dx,...,4} {
            \foreach \y in {0,\dy,...,2} {
                \node[circle, fill=black, inner sep=1pt] at (\x,\y) {};
            }
        }

        \drawroofshapex{PairedB}{0}{0}
        \drawroofshapey{PairedF}{0}{0}
