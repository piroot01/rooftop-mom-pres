\section{The problem description}
\begin{frame}{The problem description}
    \begin{itemize}
        \item Consider antenna topology $\Omega$ made from a perfect conductor.
        \item We want to know the current $\V{J}$ under exciting field $\V{E}_{\mathrm{inc}}$.
    \end{itemize}
    \vspace{1cm}
    \begin{figure}
        \centering
        \includegraphics[width=0.5\textwidth]{images/problem_descr.pdf}
    \end{figure}
\end{frame}

\begin{frame}{Equation formulation}
    \begin{itemize}
        \item The problem can be mathematically formulated using EFIE\footnotemark[1]
    \end{itemize}
    \begin{equation}
        [\V{E}_{\text{inc}}(\V{r})]_{\text{tan}} = \I\KZ\ZVAC \int\limits_{\Omega} \left(\M{I} + \frac{\nabla \otimes \nabla}{\KZ^2}\right)g(\V{r}, \V{r}')\V{J}(\V{r}')\D{\V{r}'},
    \end{equation}
    where $g$ is the 3D scalar Green's function\footnotemark[2].
    \begin{columns}
        \begin{column}[t]{0.49\textwidth}
            \centering
            \begin{minipage}{0.7\textwidth}
                \begin{block}{Scalar Green's function\footnotemark[2]}
                    \begin{equation*}
                        g(\V{r}, \V{r}') = \frac{\mathrm{e}^{-\I\KZ \|\V{r} - \V{r}'\|}}{4\uppi\|\V{r} - \V{r}'\|}
                    \end{equation*}
                \end{block}
            \end{minipage}
        \end{column}
        \begin{column}[t]{0.49\textwidth}
            \centering
            \begin{minipage}{\textwidth}
                \begin{figure}
                    \includegraphics[width=\textwidth]{images/greens_function.pdf}
                \end{figure}
            \end{minipage}
        \end{column}
    \end{columns}

    \footnotetext[1]{\fullcite{Harrington_FieldComputationByMoM}}
    \footnotetext[2]{\fullcite{Gibson_MoMinElectromagnetics}}
\end{frame}

\begin{frame}{Operators for further radiator description}
    \begin{columns}
        \begin{column}[t]{0.49\textwidth}
            \begin{block}{Far field\footnotemark[1]}
                \begin{equation*}
                    F(\vu*{r}, \vu*{e}) = \frac{-\I \ZVAC \KZ}{4 \uppi}\int_{\Omega} \vu*{e}\vdot \V{J}(\V{r}') \mathrm{e}^{\I k \vu*{r}\vdot \V{r}'} \D{\V{r}'}

\section{Solution via MoM}
\begin{frame}{Solution via Method-of-Moments paradigm}
    \begin{itemize}
        \item Upon suitable basisis we can discretize the continuous formulation via EFIE into
        \begin{equation}
            \M{Z}\V{I} = \V{V},
        \end{equation}
        where
        \begin{align}
            [\Zmat_{ij}] &= \I\KZ\ZVAC \int_{\Omega_i} \int_{\Omega_j} \PSII(\V{r}) \vdot \PSIJ(\V{r}') g(\V{r},\V{r}') \D{\V{r}'}\D{\V{r}} \\
            &\phantom{=}- \frac{\I\ZVAC}{\KZ}\int_{\Omega_i}\int_{\Omega_j}(\div{\PSII}(\V{r}))(\div{\PSIJ}(\V{r}')) g(\V{r}, \V{r}') \D{\V{r}'}\D{\V{r}},
        \end{align}
        and
        \begin{equation}
            [\V{V}_i] = \int_{\Omega_i} \PSIJ(\V{r}') \vdot \V{E}_{\text{inc}}(\V{r}') \D{\V{r}'}
        \end{equation}
    \end{itemize}
\end{frame}

\begin{frame}{Solution via Method-of-Moments paradigm}
    \begin{itemize}
        \item Then the current density $\V{J}$ can be approximated from the vector of unkwnows as
        \begin{equation}
            \V{J}(\V{r}') \approx \sum_{j = 1}^{N} [\V{I}_j] \PSIJ(\V{r'})
        \end{equation}
    \end{itemize}
\end{frame}

\section{Basis function selection}
\begin{frame}{General rooftop basis function}
    \begin{itemize}
        \item The general form of the rooftop basis functions (\textquote{rooftops}) can be expressed as
        \begin{equation}
            \PSII(\V{r}) =
            \begin{cases}
                \PSII^{+}(\V{r}) = \dfrac{l_i}{A_i^{+}}  \left((\V{r} - \V{v}_{i1}^{+})\vdot \UV{e}_i^{+}\right) \UV{e}_i^{+}, \quad & \forall \V{r} \in \Omega_i^{+}, \\[1em]
                \PSII^{-}(\V{r}) = -\dfrac{l_i}{A_i^{-}} \left( (\V{r} - \V{v}_{i2}^{-})\vdot \UV{e}_i^{-}\right) \UV{e}_i^{-}, \quad & \forall \V{r} \in \Omega_i^{-}, \\[1em]
                0, \quad & \text{elsewhere}, \\
            \end{cases}
        \end{equation}

        \end{figure}
    \end{itemize}
\end{frame}

            \item $l_i = l_y = b$ iff $\UV{e}_i^{+/-} = \UV{y}$

\begin{frame}{Singularity extraction}
    \begin{itemize}
        \item The singular scalar Green's function can be rewitten into
        \begin{equation}
            \label{eq:sigex}
            g(\V{r}, \V{r}') = g_{\text{s}}(\V{r}, \V{r}') + \frac{1}{4\uppi}R^{-1} - \frac{\KZ^2}{8\uppi}R,
        \end{equation}
        where $g_s(\V{r}, \V{r}')$, termed the smooth Green's function, is given by
        \begin{equation}
            g_{\text{s}}(\V{r}, \V{r}') = \frac{1}{4\uppi} \left(\frac{\mathrm{e}^{-\I\KZ R}}{R} -\frac{1}{R} + \frac{\KZ^2 R}{2}\right).
        \end{equation}
        \item The first term of~\eqref{eq:sigex} has a finite limit the second and third are kernels of the singular integrals.
        \item Using the transformation into the series of four line integrals we are able to obtain analytical result.
    \end{itemize}
\end{frame}

\subsection{Acceleration techniques}
\begin{frame}{Interaction symmetries approach}
    \begin{itemize}
        \item The main goal of this approach is to accelerate the impedance matrix assembly time.
        \item Each term of the impedance matrix is associated with a geomerical position of two rooftops.
        \item Identifying only those rooftops which results in a unique impedance matrix element leads to drastical reduction in construction of the impedance matrix.
        \item The translational symmetries are obvious and lead to linear time construction.

\begin{frame}{Convolutional approach}
    \begin{itemize}
        \item Reindexing the rooftops
        \begin{align}
            \V{\Psi}_{m} \longrightarrow \V{\Psi}_{\V{m}},\, \V{m} = m_x\vu*{x} + m_y\vu*{y}
        \end{align}
        then from impedance matrix is \textquote{impedance tensor} $\T{Z}$.
        \item This formulation allows us to rewrite the MoM into an equivalent set of equations which uses 2D circulant convolution.
        \item The kernels of the convolutions are denoted \textquote{Green's operator} $\mathcal{G}$ and envelopes complete information of the radiator in an irreducible form.
    \end{itemize}
\end{frame}

\begin{frame}{Pros and cons of convolutional approach}
    \begin{itemize}
        \item Because of the irreducible form of the $\mathcal{G}$, the storage and computational time to construct the MoM equations is $\mathcal{O}(N)$
        \item The goal of this approach is to primarily solve the solving time of the matrix inversion of $\mathcal{O}(N^3)$, which due to the FFT implementation goes down to $\mathcal{O}(N \log(N))$.
        \item The deficiancy of this method is current disability to solve arbitrarilu shaped radiators, the only ones are defined by the index ranges $m_x$, $m_y$.
    \end{itemize}
\end{frame}

\begin{frame}{Computation of the singular integrals}
    \begin{itemize}
        \item For demonstration purposes, we consider the following extracted integral
        \begin{equation*}
            \int_{\Omega_i}\int_{\Omega_i} \PSII^{+}(\V{r}) \vdot \PSII^{+}(\V{r}') \frac{1}{\|\V{r} - \V{r}'\|}\D{\V{r}'}\D{\V{r}}.
        \end{equation*}
        We then apply the transformations $\psi$ and $\phi$ as defined in the thesis.
    \end{itemize}
    \begin{figure}[ht]
        \centering
        \includegraphics[width=0.8\textwidth]{images/singularity_extraction.pdf}

\end{frame}

\begin{frame}
    \begin{itemize}
        \item Under the assumption of a uniform orthogonal mesh on the radiator and $a,b>0$, the integral splits into four contributions of the form
        \begin{equation*}
            a^2b^2 \int\int\int\int \frac{(u + \lambda_1')\lambda_1'}{\sqrt{a^2 u^2 + b^2 v^2}}\dd{\lambda_1'}\dd{v}\dd{u}.
        \end{equation*}
        The corresponding limits in each subregion are:
        \[
            \begin{tabu}{r l l}
                \text{I:}&\lambda_1'\in(0,1-u),&\lambda_2'\in(0,1-v), \\
                \text{II:}&\lambda_1'\in(-u,1),&\lambda_2'\in(0,1-v), \\
                \text{III:}&\lambda_1'\in(-u,1),&\lambda_2'\in(-v,1), \\
                \text{IV:}&\lambda_1'\in(0,1-u),&\lambda_2'\in(-v,1).
            \end{tabu}
        \]
        Hence
        \begin{equation*}
            \mathcal{I}^{(-1)}[\lambda_1\lambda_1'] = \mathcal{I}_{\text{I}}^{(-1)}[\lambda_1\lambda_1'] + \mathcal{I}_{\text{II}}^{(-1)}[\lambda_1\lambda_1'] + \mathcal{I}_{\text{III}}^{(-1)}[\lambda_1\lambda_1'] + \mathcal{I}_{\text{IV}}^{(-1)}[\lambda_1\lambda_1']
        \end{equation*}
    \end{itemize}
\end{frame}

\begin{frame}
    \begin{itemize}
        \item  We now compute the contribution over subregion I.
    \end{itemize}
    \begin{align*}
        \mathcal{I}_{\text{I}}^{(-1)}[\lambda_1\lambda_1'] &= a^2 b^2 \int_{0}^{1}\int_{0}^{1}\int_{0}^{1-u}\int_{0}^{1-v} \frac{(u + \lambda_1')\lambda_1'}{\sqrt{a^2 u^2 + b^2 v^2}}\D{\lambda_2'}\D{\lambda_1'}\D{v}\D{u} \\
        &= a^2 b^2 \int_{0}^{1}\int_{0}^{1} \frac{1 - v}{\sqrt{a^2 u^2 + b^2 v^2}} \int_{0}^{1-u} (u + \lambda_1') \lambda_1'\D{\lambda_1'}\D{v}\D{u} \\
        &= a^2 b^2 \int_{0}^{1}\int_{0}^{1} \frac{1 - v}{\sqrt{a^2 u^2 + b^2 v^2}} \underbrace{\left( \frac{1}{3} - \frac{1}{2}u + \frac{1}{6}u^3\right)}_{F_1(u)}\D{v} \D{v} \\
        &= a^2 b^2 \int_{0}^{1} F_1(u) \left(\int_{0}^{1}  \frac{1}{\sqrt{a^2 u^2 + b^2 v^2}} \D{v} - \int_{0}^{1}  \frac{v}{\sqrt{a^2 u^2 + b^2 v^2}} \D{v}\right) \D{u}
    \end{align*}
\end{frame}

\begin{frame}

\begin{frame}{Singularity extraction strength}
    \begin{itemize}
        \item What is the benefit of a stronger extraction?
        \item The cornerstone is the Taylor series of the scalar Green’s function,
        \begin{equation*}
            g(\V{r}, \V{r}') = \frac{1}{4\uppi}\sum_{q=-1}^{\infty} \frac{(-\I\KZ)^{q+1}}{(q+1)!}\|\V{r} - \V{r}'\|^{q},
        \end{equation*}
        \item All odd-power terms in this expansion are non-smooth at $\V{r} = \V{r}'$.
        \item We considered the first two odd terms of the expansion.
        \begin{equation*}
            g(\V{r}, \V{r}') = g_{\text{s}}(\V{r}, \V{r}') + \frac{1}{4\uppi}R^{-1} - \frac{\KZ^2}{8\uppi}R,
        \end{equation*}
        where
        \begin{equation*}
            g_{\text{s}}(\V{r}, \V{r}') = \frac{1}{4\uppi} \left(\frac{\mathrm{e}^{-\I\KZ R}}{R} -\frac{1}{R} + \frac{\KZ^2 R}{2}\right).
        \end{equation*}
        \item Integral containing $g_s$ can be evaluated numerically $\lim_{R \to 0} g_{\text{s}}(R) = -\I\KZ/(4\uppi)$ when $\V{r} = \V{r}'$.
    \end{itemize}
\end{frame}
