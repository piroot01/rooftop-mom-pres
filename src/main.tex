\section{Overview of the Problem}
\begin{frame}{Overview of the Problem}
    \begin{itemize}[<+->]
        \item Consider an antenna topology $\varOmega$, composed of a perfect conductor.
        \item Determine the current field $\V{J}\in\varOmega$ induced by the incident field $\V{E}_{\mathrm{inc}}$.
        \item The field interaction is encoded by Maxwell's equations\footnote{\fullcite{Maxwell_1865}}.
    \end{itemize}
    \vspace{0.5cm}
    \begin{figure}
        \centering
        \includegraphics[width=0.5\textwidth]{images/problem_descr.pdf}
    \end{figure}
\end{frame}

\section{Solution via MoM}
\begin{frame}{Solution via Method-of-Moments\footnotemark[1]~Paradigm}
    \begin{columns}
        \begin{column}[t]{0.49\textwidth}
            \begin{itemize}
                \item Expand the unknown current $\V{J}$ in a selected basis.
                \begin{equation*}
                    \V{J}(\V{r}') \approx \sum_{j = 1}^{N} [\M{I}_j] \boldsymbol{\psi}_j(\V{r'}),
                \end{equation*}
                then the interaction between $\V{J}$ and $\V{E}_{\text{inc}}$ is formulated as a matrix equation
                \begin{equation*}
                    \M{Z}\M{I} = \M{V} \implies \M{I} = \Zmat^{-1}\M{V}.
                \end{equation*}
            \end{itemize}
        \end{column}
        \begin{column}[t]{0.49\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=0.95\textwidth]{images/basis_function.pdf}
            \end{figure}
        \end{column}
    \end{columns}
    \footnotetext{\fullcite{Harrington_FieldComputationByMoM}}
\end{frame}

\begin{frame}{Singularity Extraction}
    \begin{itemize}
        \item The cornerstone is the Taylor series\footnote[1]{\fullcite{Handbook_of_Mathematial_functions}} of the scalar Green’s function,
        \begin{equation*}
            g(\V{r}, \V{r}') = \frac{1}{4\uppi}\sum_{q=-1}^{\infty} \frac{(-\I\KZ)^{q+1}}{(q+1)!}\|\V{r} - \V{r}'\|^{q},
        \end{equation*}
        \item All odd-power terms in this expansion are non-smooth at $\V{r} = \V{r}'$.
        \item We considered the first two odd terms of the expansion.
        \begin{equation*}
            g(\V{r}, \V{r}') = g_{\text{s}}(\V{r}, \V{r}') + \frac{1}{4\uppi}R^{-1} - \frac{\KZ^2}{8\uppi}R,
        \end{equation*}
        where $g_{\text{s}} \in C^1(\R^3 \times \R^3)$.
    \item Integrals containing $g_\text{s}$ is evaluated numerically, $\lim_{R \to 0} g_{\text{s}}(R) = -\I\KZ/(4\uppi)$ when $\V{r} = \V{r}'$.
    \end{itemize}
\end{frame}

\begin{frame}{Singular Integral Computation}
    \begin{itemize}
        \item For demonstration purposes, we consider the following extracted integral
        \begin{equation*}
            \int_{\varOmega_i}\int_{\varOmega_i} \boldsymbol{\psi}^{+}(\V{r}) \vdot \boldsymbol{\psi}^{+}(\V{r}') \frac{1}{\|\V{r} - \V{r}'\|}\D{\V{r}'}\D{\V{r}}.
        \end{equation*}
        We then apply the transformations $\psi$ and $\phi$ as defined in the thesis.
    \end{itemize}
    \begin{figure}[ht]
        \centering
        \includegraphics[width=0.8\textwidth]{images/singularity_extraction.pdf}
    \end{figure}
\end{frame}

\begin{frame}
    \begin{itemize}
        \item Under the assumption of a uniform orthogonal mesh on the radiator and $a,b>0$, the integral splits into four contributions of the form
        \begin{equation*}
            a^2b^2 \int\int\int\int \frac{(u + \lambda_1')\lambda_1'}{\sqrt{a^2 u^2 + b^2 v^2}}\dd{\lambda_1'}\dd{v}\dd{u}.
        \end{equation*}
        The corresponding limits in each subregion are:
        \[
            \begin{tabu}{r l l}
                \text{I:}&\lambda_1'\in(0,1-u),&\lambda_2'\in(0,1-v), \\
                \text{II:}&\lambda_1'\in(-u,1),&\lambda_2'\in(0,1-v), \\
                \text{III:}&\lambda_1'\in(-u,1),&\lambda_2'\in(-v,1), \\
                \text{IV:}&\lambda_1'\in(0,1-u),&\lambda_2'\in(-v,1).
            \end{tabu}
        \]
        Hence
        \begin{equation*}
            \mathcal{I}^{(-1)}[\lambda_1\lambda_1'] = \mathcal{I}_{\text{I}}^{(-1)}[\lambda_1\lambda_1'] + \mathcal{I}_{\text{II}}^{(-1)}[\lambda_1\lambda_1'] + \mathcal{I}_{\text{III}}^{(-1)}[\lambda_1\lambda_1'] + \mathcal{I}_{\text{IV}}^{(-1)}[\lambda_1\lambda_1']
        \end{equation*}
        \item A \Mathematica\ script was developed to automate the evaluation of these singular integrals.
    \end{itemize}
\end{frame}
