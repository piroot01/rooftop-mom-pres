\section{The problem description}
\begin{frame}{The problem description}
    \begin{itemize}
        \item Consider antenna topology $\Omega$, composed of a perfect conductor.
        \item We want to determine the current $\V{J}$ induced by the incident field $\V{E}_{\mathrm{inc}}$.
        \item The field interaction is encoded by Maxwell's equations.
    \end{itemize}
    \vspace{1cm}

    \end{figure}
\end{frame}

\section{Solution via MoM}
\begin{frame}{Solution via Method-of-Moments paradigm}
    \begin{itemize}
        \item Expand the unknown current $\V{J}$ in a selected basis.
        \begin{equation*}
            \V{J}(\V{r}') \approx \sum_{j = 1}^{N} [\V{I}_j] \PSIJ(\V{r'})
        \end{equation*}
        \item Then the interaction between $\V{J}$ and $\V{E}_{\text{inc}}$ is formulated as a matrix equation.
        \begin{equation*}
            \M{Z}\V{I} = \V{V},
        \end{equation*}
    \end{itemize}
    \begin{figure}
        \centering
        \includegraphics[width=0.4\textwidth]{images/basis_function.pdf}
    \end{figure}
\end{frame}

\begin{frame}
    \begin{itemize}
        \item Under the assumption of a uniform orthogonal mesh on the radiator and $a,b>0$, the integral splits into four contributions of the form
        \begin{equation*}
            a^2b^2 \int\int\int\int \frac{(u + \lambda_1')\lambda_1'}{\sqrt{a^2 u^2 + b^2 v^2}}\dd{\lambda_1'}\dd{v}\dd{u}.
        \end{equation*}
        The corresponding limits in each subregion are:
        \[
            \begin{tabu}{r l l}
                \text{I:}&\lambda_1'\in(0,1-u),&\lambda_2'\in(0,1-v), \\
                \text{II:}&\lambda_1'\in(-u,1),&\lambda_2'\in(0,1-v), \\
                \text{III:}&\lambda_1'\in(-u,1),&\lambda_2'\in(-v,1), \\
                \text{IV:}&\lambda_1'\in(0,1-u),&\lambda_2'\in(-v,1).
            \end{tabu}
        \]
        Hence
        \begin{equation*}
            \mathcal{I}^{(-1)}[\lambda_1\lambda_1'] = \mathcal{I}_{\text{I}}^{(-1)}[\lambda_1\lambda_1'] + \mathcal{I}_{\text{II}}^{(-1)}[\lambda_1\lambda_1'] + \mathcal{I}_{\text{III}}^{(-1)}[\lambda_1\lambda_1'] + \mathcal{I}_{\text{IV}}^{(-1)}[\lambda_1\lambda_1']
        \end{equation*}
    \end{itemize}
\end{frame}

\begin{frame}
    \begin{itemize}
        \item  We now compute the contribution over subregion I.
    \end{itemize}
    \begin{align*}
        \mathcal{I}_{\text{I}}^{(-1)}[\lambda_1\lambda_1'] &= a^2 b^2 \int_{0}^{1}\int_{0}^{1}\int_{0}^{1-u}\int_{0}^{1-v} \frac{(u + \lambda_1')\lambda_1'}{\sqrt{a^2 u^2 + b^2 v^2}}\D{\lambda_2'}\D{\lambda_1'}\D{v}\D{u} \\
        &= a^2 b^2 \int_{0}^{1}\int_{0}^{1} \frac{1 - v}{\sqrt{a^2 u^2 + b^2 v^2}} \int_{0}^{1-u} (u + \lambda_1') \lambda_1'\D{\lambda_1'}\D{v}\D{u} \\
        &= a^2 b^2 \int_{0}^{1}\int_{0}^{1} \frac{1 - v}{\sqrt{a^2 u^2 + b^2 v^2}} \underbrace{\left( \frac{1}{3} - \frac{1}{2}u + \frac{1}{6}u^3\right)}_{F_1(u)}\D{v} \D{v} \\
        &= a^2 b^2 \int_{0}^{1} F_1(u) \left(\int_{0}^{1}  \frac{1}{\sqrt{a^2 u^2 + b^2 v^2}} \D{v} - \int_{0}^{1}  \frac{v}{\sqrt{a^2 u^2 + b^2 v^2}} \D{v}\right) \D{u}
    \end{align*}
\end{frame}

\begin{frame}

\begin{frame}{Singularity extraction strength}
    \begin{itemize}
        \item What is the benefit of a stronger extraction?
        \item The cornerstone is the Taylor series of the scalar Green’s function,
        \begin{equation*}
            g(\V{r}, \V{r}') = \frac{1}{4\uppi}\sum_{q=-1}^{\infty} \frac{(-\I\KZ)^{q+1}}{(q+1)!}\|\V{r} - \V{r}'\|^{q},
        \end{equation*}
        \item All odd-power terms in this expansion are non-smooth at $\V{r} = \V{r}'$.
        \item We considered the first two odd terms of the expansion.
        \begin{equation*}
            g(\V{r}, \V{r}') = g_{\text{s}}(\V{r}, \V{r}') + \frac{1}{4\uppi}R^{-1} - \frac{\KZ^2}{8\uppi}R,
        \end{equation*}
        where
        \begin{equation*}
            g_{\text{s}}(\V{r}, \V{r}') = \frac{1}{4\uppi} \left(\frac{\mathrm{e}^{-\I\KZ R}}{R} -\frac{1}{R} + \frac{\KZ^2 R}{2}\right).
        \end{equation*}
        \item Integral containing $g_s$ can be evaluated numerically $\lim_{R \to 0} g_{\text{s}}(R) = -\I\KZ/(4\uppi)$ when $\V{r} = \V{r}'$.
    \end{itemize}
\end{frame}
